var replayClock atomic.Int64
func replayNow() time.Time { return time.Unix(replayClock.Load(),0).UTC() }

type shTransport struct{h http.Handler}
func (t shTransport) RoundTrip(r *http.Request) (*http.Response,error) {w:=httptest.NewRecorder();t.h.ServeHTTP(w,r);return w.Result(),nil}

func shRead(t *testing.T,path string,v any) {t.Helper();raw,err:=os.ReadFile(path);if err!=nil {t.Fatal(err)};if err=json.Unmarshal(raw,v);err!=nil {t.Fatal(err)}}
