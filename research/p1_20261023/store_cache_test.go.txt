// canonDec must equal the JSON round trip NewFromString(d.String()) in coefficient and exponent.
func TestP1CanonDecFuzz(t *testing.T) {
	r := rand.New(rand.NewSource(1))
	mk := func() decimal.Decimal {
		d := decimal.New(r.Int63n(2_000_000_000)-1_000_000_000, int32(r.Intn(24)-18))
		for k := r.Intn(4); k > 0; k-- {
			e := decimal.New(r.Int63n(1_000_000_000)+1, int32(r.Intn(16)-12))
			switch r.Intn(4) {
			case 0:
				d = d.Mul(e)
			case 1:
				d = d.Add(e)
			case 2:
				d = d.Div(e)
			default:
				d = d.Sub(e).Mul(decimal.New(10, int32(r.Intn(6)-2)))
			}
		}
		return d
	}
	for i := 0; i < 2_000_000; i++ {
		d := mk()
		c := canonDec(d)
		j, _ := decimal.NewFromString(d.String())
		if c.Exponent() != j.Exponent() || c.Coefficient().Cmp(j.Coefficient()) != 0 {
			t.Fatalf("%s: canon %v e%d json %v e%d", d.String(), c.Coefficient(), c.Exponent(), j.Coefficient(), j.Exponent())
		}
	}
}

// cloneState must equal json.Unmarshal(json.MarshalIndent(s)) field-for-field (decimals compared by coefficient+exponent).
func TestP1CloneStateMatchesJSON(t *testing.T) {
	r := rand.New(rand.NewSource(2))
	dec := func() decimal.Decimal { return decimal.New(r.Int63n(1e12), int32(r.Intn(12)-10)).Mul(decimal.New(r.Int63n(1e6)+1, -3)) }
	ts := func() time.Time { return time.Unix(1.6e9+r.Int63n(1e8), 0).UTC() }
	for i := 0; i < 2000; i++ {
		e := dec(); x := ts()
		s := State{Schema: "poloniex-bot-v1", Mode: "paper", Budget: dec(), Cash: dec(), HighWater: dec(), DayStart: dec(), Day: "2026-01-01", OrdersToday: r.Intn(5),
			Holdings: map[string]Position{"ZEC_USDT": {Quantity: dec(), Peak: dec(), Entered: ts(), Entry: &e, Exiting: "stop", ExitSince: &x}, "BNB_USDT": {Quantity: dec(), Peak: dec(), Entered: ts()}},
			Processed: map[string]bool{"a": true}, Cooldown: map[string]time.Time{"TRX_USDT": ts()}, LastCycle: ts(), LastSource: "bitbank_rotation"}
		if r.Intn(2) == 0 {
			f := dec()
			s.Pending = &Pending{Source: "x", Order: Order{Symbol: "ZEC_USDT", Side: "BUY", Price: "1.5", Quantity: "2", fillable: dec()}, Created: ts(), Fillable: &f}
		}
		for k := r.Intn(4); k > 0; k-- {
			s.Fills = append(s.Fills, Fill{Source: "y", Order: Order{Symbol: "BNB_USDT", fillable: dec()}, Quantity: dec(), Amount: dec(), Fee: dec(), At: ts(), Mode: "paper"})
		}
		if r.Intn(3) == 0 {
			s.Fills = append(s.Fills, Fill{ExchangeTrades: []Trade{}})
		}
		for k := r.Intn(5); k > 0; k-- {
			s.Equity = append(s.Equity, EquityPoint{At: ts(), Value: dec()})
		}
		if r.Intn(5) == 0 {
			s.Equity = []EquityPoint{}
		}
		b, _ := json.MarshalIndent(s, "", "  ")
		var j State
		if err := json.Unmarshal(b, &j); err != nil {
			t.Fatal(err)
		}
		c := copyState(cloneState(s))
		if !eqState(reflect.ValueOf(c), reflect.ValueOf(j)) {
			t.Fatalf("mismatch %d\n%+v\n%+v", i, c, j)
		}
	}
}

var decType = reflect.TypeOf(decimal.Decimal{})
var timeType = reflect.TypeOf(time.Time{})

func eqState(a, b reflect.Value) bool {
	if a.Type() == decType {
		x, y := a.Interface().(decimal.Decimal), b.Interface().(decimal.Decimal)
		return x.Exponent() == y.Exponent() && x.Coefficient().Cmp(y.Coefficient()) == 0
	}
	if a.Type() == timeType {
		x, y := a.Interface().(time.Time), b.Interface().(time.Time)
		return x.Equal(y) && x.Location().String() == y.Location().String() && x == y
	}
	switch a.Kind() {
	case reflect.Ptr:
		if a.IsNil() || b.IsNil() {
			return a.IsNil() == b.IsNil()
		}
		return eqState(a.Elem(), b.Elem())
	case reflect.Slice:
		if a.IsNil() != b.IsNil() || a.Len() != b.Len() {
			return false
		}
		for i := 0; i < a.Len(); i++ {
			if !eqState(a.Index(i), b.Index(i)) {
				return false
			}
		}
		return true
	case reflect.Map:
		if a.IsNil() != b.IsNil() || a.Len() != b.Len() {
			return false
		}
		for _, k := range a.MapKeys() {
			if !b.MapIndex(k).IsValid() || !eqState(a.MapIndex(k), b.MapIndex(k)) {
				return false
			}
		}
		return true
	case reflect.Struct:
		for i := 0; i < a.NumField(); i++ {
			if a.Type().Field(i).Name == "fillable" {
				x, y := a.Field(i), b.Field(i)
				_ = x
				_ = y
				continue
			}
			if !eqState(a.Field(i), b.Field(i)) {
				return false
			}
		}
		return true
	default:
		return reflect.DeepEqual(a.Interface(), b.Interface())
	}
}
