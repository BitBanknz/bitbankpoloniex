package bot

// 2026-10-11 xregime replay: the 2026-10-09 fidelity harness plus default-off entry gate, flat gate and BUY band/participation hooks.
// Book "proxy" = the 10-07 hourly proxy (one level at open -/+5 bps, depth 10% of prior-hour turnover).
// Book "archive" = a real archived Poloniex 5-level book shape per symbol, drawn deterministically per
// (minute, symbol, seed) and scaled by the mean hourly quote turnover of the prior 24 h, centred on the hour open.
// FollowCycles adds minute cycles inside an hour while any holding has a latched exit or sits at/below its
// 10% stop at the hour open (live runs every minute, so legacy stops also retry rejected sizing each minute).
// Fold-end liquidation marks holdings at the hour open times the symbol's median archived best-bid offset.

type fdHour struct { TS int64; Open, PriorTurnover, Volume24 []float64 }
type fdFixture struct { Pairs []string; Hours []fdHour; Ranks map[string][]float64; Folds [][2]int }
type fdConfig struct { Slots, Cooldown, WindowCycles, MinHoldH, ExitRamp, FollowCycles int; Reserve, HaltPeak, HaltDaily float64; TopUp bool; Budget, MaxOrder, MinExit, Book string; BookSeed uint64; Fees []float64; Gate, GateMode string; BuyBand, BuyPart float64 }
type fdSnap struct { TS int64 `json:"ts"`; B, A [][2]float64 }
var replayClock atomic.Int64
func replayNow() time.Time { return time.Unix(replayClock.Load(),0).UTC() }

type fdTransport struct{h http.Handler}
func (t fdTransport) RoundTrip(r *http.Request) (*http.Response,error) {w:=httptest.NewRecorder();t.h.ServeHTTP(w,r);return w.Result(),nil}

func fdRead(t *testing.T,path string,v any) {t.Helper();raw,err:=os.ReadFile(path);if err!=nil {t.Fatal(err)};if err=json.Unmarshal(raw,v);err!=nil {t.Fatal(err)}}
func fdMix(x uint64) uint64 {x+=0x9e3779b97f4a7c15;x=(x^(x>>30))*0xbf58476d1ce4e5b9;x=(x^(x>>27))*0x94d049bb133111eb;return x^(x>>31)}

func TestXRegimeReplay(t *testing.T) {

  defer func(){researchEntryGate=nil;researchFlatGate=nil}()

 if c.Book=="archive" {

  for i,pair:=range f.Pairs {snaps[i]=bs.Snapshots[strings.TrimSuffix(pair,"USDT")];if len(snaps[i])==0 {t.Fatalf("no archived books for %s",pair)}
   var offs []float64;for _,sn:=range snaps[i] {offs=append(offs,sn.B[0][0])};sort.Float64s(offs);liqBid[i]=1+offs[len(offs)/2]}
